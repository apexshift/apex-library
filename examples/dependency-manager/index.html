Dependency Manager

One instance, zero conflicts. Call getInstance() anywhere — you always get the same object.

📋 Config-driven

Every loaded GSAP plugin is registered with the core in the correct order.

Lenis is wired to the GSAP ticker and ScrollTrigger automatically.

📚 Implementation Guide

🚀 Basic Setup

import { DependencyManager } from 'apex/Core/DependencyManager.js';

const dm = DependencyManager.getInstance();

// init() returns a frozen map of every loaded instance
const { gsap, lenis, ScrollTrigger } = await dm.init({
  core:             ['gsap', 'lenis'],
  gsap_plugins:     ['ScrollTrigger'],
  preferredScroller: 'lenis',
  instantiate:      ['lenis'],
});

gsap.from('.hero', { y: 50, opacity: 0, duration: 1 });

⚙️ Configuration Options

await dm.init({
  // Core libraries — always loaded before plugins
  core: ['gsap', 'lenis'],

  // GSAP plugins — dependency graph is resolved automatically
  gsap_plugins: [
    'ScrollTrigger',  // requires: gsap
    'ScrollSmoother', // requires: gsap, ScrollTrigger
    'CustomEase',     // requires: gsap
  ],

  preferredScroller: 'lenis',   // 'lenis' | 'ScrollSmoother'
  instantiate: ['lenis'],

  lenisConfig: {
    duration:    1.2,
    easing:      'outCubic',
    smoothWheel: true,
  },

  // Opt-in: exposes deps at window.Apex.deps
  exposeGlobal: false,
});

📡 Event Handling

dm.on('init:start',        () => console.log('Loading…'));
dm.on('dep:loaded',    (d) => console.log(`${d.name} loaded`));
dm.on('plugin:registered', (d) => console.log(`${d.name} registered`));
dm.on('error',         (d) => console.error(d.name, d.error));

dm.on('ready', () => {
  // dm.loaded is a frozen snapshot of all loaded instances
  const { gsap, ScrollTrigger } = dm.loaded;
  gsap.registerPlugin(ScrollTrigger);
});

🎨 Real-world Usage

async function initApp() {
  const dm = DependencyManager.getInstance();

  const { gsap, lenis, ScrollTrigger } = await dm.init({
    core:             ['gsap', 'lenis'],
    gsap_plugins:     ['ScrollTrigger'],
    preferredScroller: 'lenis',
    instantiate:      ['lenis'],
  });

  gsap.from('.hero-title', {
    y: 50, opacity: 0, duration: 1, ease: 'power2.out',
  });

  ScrollTrigger.create({
    trigger: '.section',
    start:   'top center',
    onEnter: () => console.log('entered'),
  });

  lenis.on('scroll', ({ scroll }) => console.log(scroll));
}

initApp();

🔄 Advanced Patterns

// Per-page config
const pageConfig = {
  home:    { core: ['gsap', 'lenis'], gsap_plugins: ['ScrollTrigger'] },
  contact: { core: ['gsap'],          gsap_plugins: ['CustomEase'] },
};

const page = location.pathname.split('/').pop() || 'home';
const deps = await dm.init(pageConfig[page] ?? pageConfig.home);

// On-demand plugin loading
async function loadPlugin(name) {
  if (!dm.loaded[name]) await dm.init({ gsap_plugins: [name] });
  return dm.loaded[name];
}

// Error handling
dm.on('error', ({ name, error }) => {
  console.error(`Failed to load ${name}`, error);
});

React

Custom hook pattern with cleanup

Create a useDependencyManager hook. Call it once at the app root — the singleton ensures all child components share the same loaded state.

// hooks/useDependencyManager.js
import { useEffect, useState } from 'react';
import { DependencyManager } from 'apex/Core/DependencyManager.js';

export function useDependencyManager(config) {
  const [deps, setDeps]   = useState({});
  const [error, setError] = useState(null);

  useEffect(() => {
    const dm = DependencyManager.getInstance();
    dm.on('error', (d) => setError(d.error));
    dm.init(config).then(setDeps).catch(setError);
  }, []);

  return { isReady: Object.keys(deps).length > 0, deps, error };
}

// App.jsx
export default function App() {
  const { isReady, deps } = useDependencyManager({
    core:             ['gsap', 'lenis'],
    gsap_plugins:     ['ScrollTrigger'],
    preferredScroller: 'lenis',
    instantiate:      ['lenis'],
  });

  if (!isReady) return <div>Loading…</div>;
  return <main><HeroSection gsap={deps.gsap} /></main>;
}

Vue 3

Composable + app-level plugin

Expose the DependencyManager as a Vue plugin installed in main.js, then consume via useApex() anywhere in your component tree.

// plugins/apex.js
import { DependencyManager } from 'apex/Core/DependencyManager.js';
import { ref, inject }       from 'vue';

const APEX_KEY = Symbol('apex');

export const ApexPlugin = {
  install(app, config = {}) {
    const dm     = DependencyManager.getInstance();
    const deps   = ref({});
    dm.init(config).then((d) => { deps.value = d; });
    app.provide(APEX_KEY, { dm, deps });
  },
};

export function useApex() { return inject(APEX_KEY); }

// main.js
createApp(App)
  .use(ApexPlugin, {
    core:             ['gsap', 'lenis'],
    gsap_plugins:     ['ScrollTrigger'],
    preferredScroller: 'lenis',
    instantiate:      ['lenis'],
  })
  .mount('#app');

Shopify

Dawn / OS 2.0 themes — theme.liquid + asset file

Upload the compiled DependencyManager.js to your theme's Assets folder and reference it via the Liquid asset_url filter.

<!-- theme.liquid — before </body> -->
<script type="module">
  import { DependencyManager } from "{{ 'DependencyManager.js' | asset_url }}";

  const dm = DependencyManager.getInstance();

  dm.on('ready', () => {
    const { gsap, lenis } = dm.loaded;
    document.dispatchEvent(new CustomEvent('apex:ready', { detail: dm.loaded }));
  });

  dm.init({ core: ['gsap', 'lenis'], gsap_plugins: ['ScrollTrigger'],
            preferredScroller: 'lenis', instantiate: ['lenis'] });
</script>

<!-- In any section/snippet -->
<script type="module">
  document.addEventListener('apex:ready', ({ detail }) => {
    detail.gsap.from('.hero__title', { y: 40, opacity: 0, duration: 1 });
  });
</script>

WordPress

Child theme via wp_enqueue_scripts

WordPress doesn't natively output type="module" — use a script_loader_tag filter to add it.

// child-theme/functions.php
add_action( 'wp_enqueue_scripts', function() {
  wp_enqueue_script( 'apex-dm',
    get_stylesheet_directory_uri() . '/js/DependencyManager.js',
    [], null, [ 'strategy' => 'defer', 'in_footer' => true ] );

  wp_enqueue_script( 'apex-init',
    get_stylesheet_directory_uri() . '/js/apex-init.js',
    [ 'apex-dm' ], null, [ 'strategy' => 'defer', 'in_footer' => true ] );
} );

add_filter( 'script_loader_tag', function( $tag, $handle ) {
  if ( in_array( $handle, [ 'apex-dm', 'apex-init' ] ) ) {
    return str_replace( ' src=', ' type="module" src=', $tag );
  }
  return $tag;
}, 10, 2 );

Bricks Builder

Child theme — init after Bricks frontend scripts load

Hook into wp_enqueue_scripts with priority 99 and wait for the bricksReady event so all Bricks elements are in the DOM.

// Same enqueue + filter as WordPress above (priority 99)
add_action( 'wp_enqueue_scripts', function() { /* … */ }, 99 );

// js/apex-init.js
import { DependencyManager } from './DependencyManager.js';

async function init() {
  const dm = DependencyManager.getInstance();

  const { gsap, ScrollTrigger } = await dm.init({
    core:             ['gsap', 'lenis'],
    gsap_plugins:     ['ScrollTrigger'],
    preferredScroller: 'lenis',
    instantiate:      ['lenis'],
  });

  gsap.from('[data-apex-animate]', {
    y: 40, opacity: 0, stagger: 0.12,
    scrollTrigger: { trigger: '[data-apex-animate]', start: 'top 85%' },
  });
}

// bricksReady fires once Bricks has mounted all elements
document.addEventListener('bricksReady', init);

Webflow

Custom code panel — no build step required

Go to Project Settings → Custom Code → Footer Code. Host the compiled JS on a CDN and import it as a module.

Note: Webflow injects its own Lenis on some plans. Set preferredScroller: 'lenis' — the conflict resolver detects and defers to Webflow's instance automatically. 
<!-- Project Settings → Custom Code → Footer Code -->
<script type="module">
  import { DependencyManager }
    from 'https://cdn.yourdomain.com/apex/DependencyManager.js';

  const dm = DependencyManager.getInstance();

  dm.on('ready', () => {
    const { gsap, ScrollTrigger } = dm.loaded;
    gsap.utils.toArray('.cms-item').forEach((el, i) => {
      gsap.from(el, {
        y: 30, opacity: 0, duration: 0.8, delay: i * 0.1,
        scrollTrigger: { trigger: el, start: 'top 88%' },
      });
    });
  });

  // Webflow.push defers init until after IX2
  window.Webflow ||= [];
  window.Webflow.push(() => {
    dm.init({ core: ['gsap'], gsap_plugins: ['ScrollTrigger', 'CustomEase'],
              preferredScroller: 'lenis' });
  });
</script>
