📚 Implementation Guide

Learn how to integrate the DependencyManager into your projects

🚀 Basic Setup

// Import the DependencyManager
import { DependencyManager } from './path/to/DependencyManager.js';

// Get the singleton instance
const dm = DependencyManager.getInstance();

// Initialize with your dependencies
await dm.init({
  core: ['gsap'],           // Core libraries
  gsap_plugins: ['ScrollTrigger', 'ScrollSmoother'], // GSAP plugins
  preferredScroller: 'lenis', // 'lenis' or 'ScrollSmoother'
  instantiate: ['lenis']    // Auto-instantiate these modules
});

// Your dependencies are now available globally!
console.log(window.Apex.deps); // { gsap: {...}, ScrollTrigger: {...}, lenis: {...} }

⚙️ Configuration Options

await dm.init({
  // Core dependencies (always loaded first)
  core: ['gsap', 'lenis'],

  // GSAP plugins (auto-resolves dependencies)
  gsap_plugins: [
    'ScrollTrigger',    // Requires: gsap
    'ScrollSmoother',   // Requires: gsap, ScrollTrigger
    'CustomEase'        // Requires: gsap
  ],

  // Scroll library preference
  preferredScroller: 'lenis', // or 'ScrollSmoother'

  // Auto-instantiate these modules
  instantiate: ['lenis'],

  // Custom Lenis configuration
  lenisConfig: {
    duration: 1.2,
    easing: 'outCubic',
    smoothWheel: true
  }
});

📡 Event Handling

dm.on('init:start', () => console.log('🚀 Loading dependencies...'));

dm.on('dep:loaded', (data) => console.log(`✅ ${data.name} loaded`));

dm.on('plugin:registered', (data) => console.log(`🔧 ${data.name} registered with GSAP`));

dm.on('ready', () => {
  console.log('🎉 All dependencies ready!');
  const { gsap, ScrollTrigger } = window.Apex.deps;
});

if (dm.isReady) {
  const loadedDeps = dm.loaded;
}

🎨 Real-world Usage

async function initApp() {
  const dm = DependencyManager.getInstance();

  try {
    await dm.init({
      core: ['gsap'],
      gsap_plugins: ['ScrollTrigger', 'ScrollSmoother'],
      preferredScroller: 'lenis',
      instantiate: ['lenis']
    });

    initAnimations();
    initScrollEffects();
  } catch (error) {
    console.error('Failed to load dependencies:', error);
  }
}

function initAnimations() {
  const { gsap, ScrollTrigger } = window.Apex.deps;

  gsap.from('.hero-title', { y: 50, opacity: 0, duration: 1, ease: 'power2.out' });

  ScrollTrigger.create({
    trigger: '.section',
    start: 'top center',
    onEnter: () => console.log('Section entered viewport')
  });
}

function initScrollEffects() {
  const lenis = window.Apex.deps.lenis;
  lenis.on('scroll', (data) => { /* handle scroll */ });
}

initApp();

🔄 Advanced Patterns

// Conditional loading based on page
const pageConfig = {
  home: { core: ['gsap', 'lenis'], gsap_plugins: ['ScrollTrigger', 'ScrollSmoother'] },
  about: { core: ['gsap'], gsap_plugins: ['ScrollTrigger'] },
  contact: { core: ['gsap'], gsap_plugins: ['CustomEase'] }
};

const currentPage = window.location.pathname.split('/').pop() || 'home';
await dm.init(pageConfig[currentPage] || pageConfig.home);

// Dynamic plugin loading
async function loadPluginOnDemand(pluginName) {
  if (!dm.loaded[pluginName]) {
    await dm.init({ gsap_plugins: [pluginName] });
  }
  return dm.loaded[pluginName];
}

// Error handling
dm.on('error', (data) => {
  console.error(`Failed to load ${data.name}:`, data.error);
  if (data.name === 'lenis') console.log('Falling back to native scroll');
});

React

Custom hook pattern with cleanup

Create a reusable useDependencyManager hook. Call it once at the app root — the singleton ensures all child components share the same loaded state.

// hooks/useDependencyManager.js
import { useEffect, useState } from 'react';
import { DependencyManager } from '@apex/library';

export function useDependencyManager(config) {
  const [isReady, setIsReady] = useState(false);
  const [error, setError] = useState(null);

  useEffect(() => {
    const dm = DependencyManager.getInstance();
    dm.on('ready', () => setIsReady(true));
    dm.on('error', (data) => setError(data.error));
    dm.init(config).catch(setError);
  }, []);

  return { isReady, error, deps: window.Apex?.deps ?? {} };
}

// App.jsx
export default function App() {
  const { isReady, deps } = useDependencyManager({
    core: ['gsap', 'lenis'],
    gsap_plugins: ['ScrollTrigger'],
    preferredScroller: 'lenis',
    instantiate: ['lenis'],
  });

  if (!isReady) return <div>Loading...</div>;
  return <main><HeroSection gsap={deps.gsap} /></main>;
}

Vue 3

Composable + app-level plugin

Expose the DependencyManager as a Vue plugin installed in main.js, then consume via a useApex() composable anywhere in your component tree.

// plugins/apex.js
import { DependencyManager } from '@apex/library';
import { ref, inject } from 'vue';

const APEX_KEY = Symbol('apex');

export const ApexPlugin = {
  install(app, config = {}) {
    const isReady = ref(false);
    const dm = DependencyManager.getInstance();
    dm.on('ready', () => { isReady.value = true; });
    dm.init(config);
    app.provide(APEX_KEY, { dm, isReady });
  },
};

export function useApex() { return inject(APEX_KEY); }

// main.js
createApp(App)
  .use(ApexPlugin, {
    core: ['gsap', 'lenis'],
    gsap_plugins: ['ScrollTrigger'],
    preferredScroller: 'lenis',
    instantiate: ['lenis'],
  })
  .mount('#app');

Shopify

Dawn / OS 2.0 themes — theme.liquid + asset file

Upload the compiled DependencyManager.js to your theme's Assets folder and reference it via the Liquid asset_url filter.

<!-- theme.liquid — just before </body> -->
<script type="module" src="{{ 'DependencyManager.js' | asset_url }}" defer></script>

<script type="module">
  import { DependencyManager } from "{{ 'DependencyManager.js' | asset_url }}";
  const dm = DependencyManager.getInstance();

  dm.on('ready', () => {
    document.dispatchEvent(new CustomEvent('apex:ready', { detail: window.Apex.deps }));
  });

  dm.init({ core: ['gsap', 'lenis'], gsap_plugins: ['ScrollTrigger'],
            preferredScroller: 'lenis', instantiate: ['lenis'] });
</script>

<!-- In any section/snippet -->
<script type="module">
  document.addEventListener('apex:ready', ({ detail }) => {
    detail.gsap.from('.hero__title', { y: 40, opacity: 0, duration: 1 });
  });
</script>

WordPress

Child theme via wp_enqueue_scripts

WordPress doesn't natively output type="module" — use a script_loader_tag filter to add it.

// child-theme/functions.php
add_action( 'wp_enqueue_scripts', function() {
  wp_enqueue_script( 'apex-dm',
    get_stylesheet_directory_uri() . '/js/DependencyManager.js',
    [], null, [ 'strategy' => 'defer', 'in_footer' => true ] );

  wp_enqueue_script( 'apex-init',
    get_stylesheet_directory_uri() . '/js/apex-init.js',
    [ 'apex-dm' ], null, [ 'strategy' => 'defer', 'in_footer' => true ] );
} );

add_filter( 'script_loader_tag', function( $tag, $handle ) {
  if ( in_array( $handle, [ 'apex-dm', 'apex-init' ] ) ) {
    return str_replace( ' src=', ' type="module" src=', $tag );
  }
  return $tag;
}, 10, 2 );

Bricks Builder

Child theme — init after Bricks frontend scripts load

Hook into wp_enqueue_scripts with priority 99 and wait for the bricksReady event so all Bricks elements are in the DOM.

// Same enqueue + filter as WordPress above, with priority 99
add_action( 'wp_enqueue_scripts', function() { /* ... */ }, 99 );

// js/apex-init.js
import { DependencyManager } from './DependencyManager.js';

async function init() {
  const dm = DependencyManager.getInstance();
  await dm.init({ core: ['gsap', 'lenis'], gsap_plugins: ['ScrollTrigger'],
                  preferredScroller: 'lenis', instantiate: ['lenis'] });

  const { gsap } = window.Apex.deps;
  gsap.from('[data-apex-animate]', {
    y: 40, opacity: 0, stagger: 0.12,
    scrollTrigger: { trigger: '[data-apex-animate]', start: 'top 85%' },
  });
}

// bricksReady fires once Bricks has mounted all elements
document.addEventListener('bricksReady', init);

Webflow

Custom code panel — no build step required

Go to Project Settings → Custom Code → Footer Code. Host the compiled JS on a CDN and import it as a module.

⚠️ Webflow note: Webflow injects its own Lenis instance on some plans. Set preferredScroller: 'lenis' — the conflict resolver will detect and defer to Webflow's instance automatically. 
<!-- Project Settings → Custom Code → Footer Code -->
<script type="module">
  import { DependencyManager } from 'https://cdn.yourdomain.com/apex/DependencyManager.js';

  const dm = DependencyManager.getInstance();

  dm.on('ready', () => {
    const { gsap, ScrollTrigger } = window.Apex.deps;
    gsap.utils.toArray('.cms-item').forEach((el, i) => {
      gsap.from(el, { y: 30, opacity: 0, duration: 0.8, delay: i * 0.1,
                      scrollTrigger: { trigger: el, start: 'top 88%' } });
    });
  });

  // Webflow.push defers init until after IX2
  window.Webflow ||= [];
  window.Webflow.push(() => {
    dm.init({ core: ['gsap'], gsap_plugins: ['ScrollTrigger', 'CustomEase'],
              preferredScroller: 'lenis' });
  });
</script>
